scipy.optimize.curve_fit — ajustar curvas

Cuando tienes datos experimentales y quieres ajustar un modelo:

from scipy.optimize import curve_fit

# 1. Definir la función del modelo
def modelo_lineal(x, a, b):
    return a * x + b

# 2. Datos experimentales
x_data = np.array([0, 1, 2, 3, 4, 5])
y_data = np.array([0.1, 1.9, 4.2, 5.8, 8.1, 9.9])

# 3. Ajustar — curve_fit devuelve parámetros óptimos
params, cov = curve_fit(modelo_lineal, x_data, y_data)
a, b = params
print(f"Ecuación: y = {a:.3f}x + {b:.3f}")
Ecuación: y = 1.980x + 0.114

curve_fit aplicado a datos agronómicos

# Modelo de respuesta de rendimiento al agua (FAO-33)
def fao33_yield_response(ETa_ETm, Ky):
    """Ky: factor de respuesta de rendimiento"""
    return 1 - Ky * (1 - ETa_ETm)

# Datos: fracción ETa/ETm vs rendimiento relativo
eta_ratio = np.array([1.0, 0.85, 0.70, 0.55, 0.40])
yield_rel = np.array([1.0, 0.82, 0.60, 0.38, 0.15])

# Ajustar Ky
params, cov = curve_fit(fao33_yield_response, eta_ratio, yield_rel)
Ky = params[0]
print(f"Factor de respuesta Ky = {Ky:.2f}")
# Visualizar ajuste
x_smooth = np.linspace(0.3, 1.0, 100)
y_fit = fao33_yield_response(x_smooth, Ky)

ax.plot(x_smooth, y_fit, 'r-', label=f'Modelo FAO-33 (Ky={Ky:.2f})')
ax.scatter(eta_ratio, yield_rel, s=80, label='Datos')

Funciones personalizadas con lambda

# lambda: función en una línea
cuadrado = lambda x: x ** 2
cuadrado(5)    # 25

# Muy útil con apply()
datos['T_C'] = [22, 18, 30, 15]
datos['T_K'] = datos['T_C'].apply(lambda x: x + 273.15)
# Aplicar una función a cada fila
def calc_Dpv(T, HR):
    """Déficit de presión de vapor"""
    es = 0.6108 * np.exp(17.27 * T / (T + 237.3))
    ea = es * HR / 100
    return es - ea

# apply con axis=1 (por fila)
datos['DPV'] = datos.apply(
    lambda row: calc_Dpv(row['T_C'], row['HR_pct']),
    axis=1
)

Ecuaciones de interés agronómico en Python

import numpy as np

# Presión de vapor de saturación (Tetens)
def es(T_C):
    return 0.6108 * np.exp(17.27 * T_C / (T_C + 237.3))

# Déficit de presión de vapor
def dpv(T_C, HR_pct):
    ea = es(T_C) * HR_pct / 100
    return es(T_C) - ea

# Conductancia estomática relativa
def gs_rel(psi_leaf, psi_50, slope):
    """Modelo sigmoide de respuesta estomática"""
    return 1 / (1 + np.exp(slope * (psi_leaf - psi_50)))

Tip

Encapsular fórmulas en funciones evita repetir código y reduce errores.
