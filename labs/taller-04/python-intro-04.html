import pandas as pd

# Crear un rango de fechas

Calcular variables derivadas

# Temperatura media
clima['Tmed_C'] = (clima['Tmax_C'] + clima['Tmin_C']) / 2

# Déficit de presión de vapor
def calc_es(T_C):
    """Presión de vapor de saturación (kPa) — ecuación de Tetens"""
    return 0.6108 * np.exp(17.27 * T_C / (T_C + 237.3))

es_Tmax = calc_es(clima['Tmax_C'])
es_Tmin = calc_es(clima['Tmin_C'])
clima['es_kPa'] = (es_Tmax + es_Tmin) / 2
clima['ea_kPa'] = clima['es_kPa'] * clima['HR_pct'] / 100
clima['DPV_kPa'] = clima['es_kPa'] - clima['ea_kPa']

Ecuación de Hargreaves en Python

\[ETo = 0.0023 \cdot R_a \cdot (T_{med} + 17.8) \cdot \sqrt{T_{max} - T_{min}}\]

def hargreaves_ETo(Tmax, Tmin, Ra, lat_rad):
    """
    Calcula ETo por Hargreaves-Samani.
    Tmax, Tmin en °C
    Ra en MJ/m²/día (radiación extraterrestre)
    Retorna ETo en mm/día
    """
    Tmed = (Tmax + Tmin) / 2
    ETo = 0.0023 * Ra * (Tmed + 17.8) * np.sqrt(Tmax - Tmin)
    return ETo

# Aplicar a todo el DataFrame
clima['ETo_Hargreaves_mm'] = hargreaves_ETo(
    clima['Tmax_C'],
    clima['Tmin_C'],
    clima['Ra_MJ_m2'],
    lat_rad=-0.589  # Santiago ≈ 33.5°S en radianes
)

Coeficientes de cultivo (Kc)

La ETc se calcula como: \(ET_c = K_c \times ETo\)

# Kc por etapa fenológica para maíz (FAO-56)
kc_maiz = pd.DataFrame({
    'Etapa': ['Inicial', 'Desarrollo', 'Media', 'Final'],
    'Dias': [25, 40, 50, 30],
    'Kc': [0.30, 0.75, 1.20, 0.55]
})

kc_maiz['Dia_acumulado'] = kc_maiz['Dias'].cumsum()

# Interpolar Kc diario
dias = np.arange(1, kc_maiz['Dia_acumulado'].iloc[-1] + 1)
kc_diario = np.interp(dias,
    kc_maiz['Dia_acumulado'] - kc_maiz['Dias'] + 1,
    kc_maiz['Kc'])

# Calcular ETc
clima['Kc'] = kc_diario[clima.index.dayofyear - 1]
clima['ETc_mm'] = clima['ETo_Hargreaves_mm'] * clima['Kc']

np.interp() — interpolación lineal simple

Alternativa más ligera que scipy.interpolate:

import numpy as np

# Puntos conocidos
x_known = np.array([0, 10, 20, 30])
y_known = np.array([0.3, 0.7, 1.1, 0.6])

# Interpolar en nuevos puntos
x_new = np.array([5, 12, 25])
y_new = np.interp(x_new, x_known, y_known)

print(f"Kc interpolados: {y_new}")
Kc interpolados: [0.5  0.74  0.85]

Tip

np.interp() es más rápido que interp1d para interpolación lineal simple.

cumsum() — sumas acumuladas

# Calcular días acumulados por etapa
kc_maiz['Dia_acumulado'] = kc_maiz['Dias'].cumsum()

# Calcular ETo acumulada en la temporada
clima['ETo_acum_mm'] = clima['ETo_Hargreaves_mm'].cumsum()

# Calcular ETc acumulada
clima['ETc_acum_mm'] = clima['ETc_mm'].cumsum()
# Visualizar acumulados
fig, ax = plt.subplots(figsize=(12, 5))

        'b-', linewidth=2, label='ETo acumulada')

        'g-', linewidth=2, label='ETc acumulada')
